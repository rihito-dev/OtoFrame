cmake_minimum_required(VERSION 3.16)
project(OtoFrame VERSION 0.1.0 LANGUAGES CXX)

option(OTOFRAME_PORTABLE "Use portable FFT even on Apple platforms" OFF)
add_library(OtoFrame INTERFACE)
add_library(OtoFrame::OtoFrame ALIAS OtoFrame)
target_include_directories(OtoFrame INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(OtoFrame INTERFACE cxx_std_17)
if(APPLE AND NOT OTOFRAME_PORTABLE)
    target_link_libraries(OtoFrame INTERFACE "-framework Accelerate")
else()
    target_compile_definitions(OtoFrame INTERFACE OTOFRAME_PORTABLE=1)
endif()

add_executable(otoframe examples/analyze.cpp)
target_link_libraries(otoframe PRIVATE OtoFrame)
include(CTest)
if(BUILD_TESTING)
    add_executable(features-tests tests/features_test.cpp)
    target_link_libraries(features-tests PRIVATE OtoFrame)
    add_test(NAME features COMMAND features-tests)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_test.py $<TARGET_FILE:otoframe>)
    endif()
endif()
